  <script>
    let dataUserGlobal = null; 

    document.getElementById('loginForm').addEventListener('submit', function(e) {
      e.preventDefault(); 
      const emailInput = document.getElementById('email').value;
      const passInput = document.getElementById('password').value;
      const btnSubmit = document.getElementById('btnSubmit');
      
      btnSubmit.innerHTML = '<span class="spinner-border spinner-border-sm me-2" role="status" aria-hidden="true"></span>Memeriksa...'; 
      btnSubmit.disabled = true;

      // === GANTI "URL_API_MAS_BRO" DENGAN LINK DEPLOY LANGKAH 2 ===
      const API_URL = "URL_API_MAS_BRO";

      // Mengirim data ke Mesin Google tanpa menggunakan google.script.run
      fetch(API_URL, {
        method: "POST",
        headers: {
          "Content-Type": "text/plain;charset=utf-8", 
        },
        body: JSON.stringify({ email: emailInput, password: passInput })
      })
      .then(response => response.json())
      .then(respon => {
        if (respon.status === 'success') {
          tampilkanDashboard(respon);
        } else {
          alert(respon.pesan);
          btnSubmit.innerHTML = 'Masuk Aplikasi'; 
          btnSubmit.disabled = false;
        }
      })
      .catch(error => {
        alert("Gagal menghubungi server: " + error.message);
        btnSubmit.innerHTML = 'Masuk Aplikasi'; 
        btnSubmit.disabled = false;
      });

    function tampilkanDashboard(data) {
      dataUserGlobal = data;
      const warnaTema = data.tenant.warnaTema || '#003366';
      
      // Mengubah Variabel CSS agar semua warna elemen ikut berubah
      document.documentElement.style.setProperty('--warna-tema', warnaTema);
      
      document.getElementById('halamanLogin').style.display = 'none';
      document.getElementById('halamanDashboard').style.display = 'block';
      
      // Suntik Data Teks
      document.getElementById('namaSekolahNav').innerHTML = `<i class="bi bi-buildings me-2"></i>` + data.tenant.namaSekolah;
      document.getElementById('namaUserLengkap').innerText = data.user.nama;
      document.getElementById('roleUser').innerText = data.user.role;

      // Panggil pembuat menu
      renderMenu(data.user.role);
    }

    // --- FUNGSI MERAKIT MENU DENGAN BOOTSTRAP ICONS ELEGANT ---
    function renderMenu(role) {
      const areaMenu = document.getElementById('areaMenu');
      let htmlMenu = '';
      
      // Menggunakan ikon garis murni (outline) dari Bootstrap Icons untuk kesan elegan
      if (role.toLowerCase() === 'guru' || role.toLowerCase() === 'pendidik') {
        htmlMenu += buatKotakMenu('bi-person-check', 'Presensi Kelas', 'bukaModul("presensi")', 1);
        htmlMenu += buatKotakMenu('bi-award', 'Input Nilai', 'bukaModul("nilai")', 2);
        htmlMenu += buatKotakMenu('bi-book', 'E-Jurnal', 'bukaModul("jurnal")', 3);
        htmlMenu += buatKotakMenu('bi-calendar3-week', 'Jadwal Anda', 'bukaModul("jadwal")', 4);
      } 
      else if (role.toLowerCase() === 'admin sekolah' || role.toLowerCase() === 'admin') {
        htmlMenu += buatKotakMenu('bi-people', 'Data Siswa', 'bukaModul("siswa")', 1);
        htmlMenu += buatKotakMenu('bi-person-vcard', 'Data Guru', 'bukaModul("guru")', 2);
        htmlMenu += buatKotakMenu('bi-arrow-left-right', 'Mutasi', 'bukaModul("mutasi")', 3);
        htmlMenu += buatKotakMenu('bi-envelope-paper', 'Persuratan', 'bukaModul("surat")', 4);
      }
      else if (role.toLowerCase() === 'siswa' || role.toLowerCase() === 'orang tua') {
        htmlMenu += buatKotakMenu('bi-bar-chart-line', 'Rapor Akademik', 'bukaModul("rapor")', 1);
        htmlMenu += buatKotakMenu('bi-wallet2', 'Tagihan SPP', 'bukaModul("tagihan")', 2);
        htmlMenu += buatKotakMenu('bi-calendar-check', 'Cek Absensi', 'bukaModul("cek_absen")', 3);
      }
      else {
        // Default Super Admin / Yayasan
        htmlMenu += buatKotakMenu('bi-pie-chart', 'Statistik', 'bukaModul("statistik")', 1);
        htmlMenu += buatKotakMenu('bi-gear', 'Pengaturan', 'bukaModul("setting")', 2);
      }

      areaMenu.innerHTML = htmlMenu;
    }

    // Fungsi pembuat Kotak HTML dengan Delay Animasi
    function buatKotakMenu(iconClass, judul, aksi, index) {
      // Delay bertingkat (0.1s, 0.2s, 0.3s dst) untuk efek muncul berurutan
      const delay = index * 0.1; 
      
      return `
        <div class="menu-card" onclick="${aksi}" style="animation: slideUpFade 0.5s ease forwards ${delay}s;">
          <div class="menu-icon-wrap">
            <i class="bi ${iconClass}"></i>
          </div>
          <p class="menu-title">${judul}</p>
        </div>
      `;
    }

    function bukaModul(namaModul) {
      // Efek getar (haptic feedback) untuk HP jika didukung
      if (navigator.vibrate) navigator.vibrate(50);
      alert("Aplikasi akan memuat modul: " + namaModul.toUpperCase());
    }

    function logout() { window.top.location.reload(); }
  </script>
